<ParamField body="pricing" type="object">
  Prices per model id, as `[pricing."<model>"]`, in USD per million tokens. Without them, usage is reported in tokens only.
</ParamField>

<ParamField body="pricing.<name>.input" type="number" required>
  Input tokens.
</ParamField>

<ParamField body="pricing.<name>.output" type="number" required>
  Output tokens.
</ParamField>

<ParamField body="pricing.<name>.cache_write" type="number">
  Cache-write tokens.
</ParamField>

<ParamField body="pricing.<name>.cache_read" type="number">
  Cache-read tokens.
</ParamField>

<ParamField body="pricing.<name>.tiers" type="object[]">
  Higher rates for a call whose prompt is over a token count, as `[[pricing."<model>".tiers]]`. The prompt counts input, cache-write and cache-read tokens. The highest tier the prompt passes prices every token of the call. A tier's missing output rate keeps the base rate; its missing cache rates scale with its input rate.
</ParamField>

<ParamField body="pricing.<name>.tiers.[].above" type="integer" required>
  The tier applies when the prompt has more tokens than this.
</ParamField>

<ParamField body="pricing.<name>.tiers.[].input" type="number">
  Input tokens.
</ParamField>

<ParamField body="pricing.<name>.tiers.[].output" type="number">
  Output tokens.
</ParamField>

<ParamField body="pricing.<name>.tiers.[].cache_write" type="number">
  Cache-write tokens.
</ParamField>

<ParamField body="pricing.<name>.tiers.[].cache_read" type="number">
  Cache-read tokens.
</ParamField>
