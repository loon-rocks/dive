<ParamField body="mcp" type="object">
  MCP servers, as `[mcp.<name>]`. Their tools join the catalog as `<name>__<tool>`.
</ParamField>

<ParamField body="mcp.<name>.description" type="string">
  What the server is for, read by drafters choosing tools.
</ParamField>

<ParamField body="mcp.<name>.command" type="string">
  Executable for a stdio server. Use `command` or `url`, not both.
</ParamField>

<ParamField body="mcp.<name>.args" type="string[]">
  Arguments for `command`.
</ParamField>

<ParamField body="mcp.<name>.env" type="map<string, string>">
  Environment for a stdio server. Values expand `${NAME}` references.
</ParamField>

<ParamField body="mcp.<name>.url" type="string">
  Endpoint for a streamable HTTP server.
</ParamField>

<ParamField body="mcp.<name>.headers" type="map<string, string>">
  HTTP headers. Values expand `${NAME}` references.
</ParamField>

<ParamField body="mcp.<name>.include_tools" type="string[]">
  Load only these tools.
</ParamField>

<ParamField body="mcp.<name>.exclude_tools" type="string[]">
  Skip these tools.
</ParamField>

<ParamField body="mcp.<name>.files" type="enum<string>" default="path">
  How a file reference (`blob:sha256-…`) in a tool call's arguments reaches the server: `path` as a local file path, `url` as a link the server can fetch, `base64` as the file's contents, `off` as the reference text unchanged.

  Available options: `path`, `url`, `base64`, `off`
</ParamField>

<ParamField body="mcp.<name>.oauth" type="object">
  The OAuth app Dive signs in with, as `[mcp.<name>.oauth]`, for servers whose authorization server doesn't let apps register themselves. Register the app with the callback URL `http://127.0.0.1/callback`.
</ParamField>

<ParamField body="mcp.<name>.oauth.client_id" type="string" required>
  The app's client ID.
</ParamField>

<ParamField body="mcp.<name>.oauth.client_secret" type="string">
  The app's client secret. Keep it out of the file with a `${NAME}` reference.
</ParamField>

<ParamField body="mcp.<name>.oauth.scopes" type="string[]">
  The access to ask for. Without it, Dive asks for what the server lists.
</ParamField>

<ParamField body="mcp.<name>.tool_overrides" type="object">
  Per-tool corrections, as `[mcp.<name>.tool_overrides.<tool>]`.
</ParamField>

<ParamField body="mcp.<name>.tool_overrides.<name>.description" type="string">
  Replaces the server's description of the tool.
</ParamField>

<ParamField body="mcp.<name>.tool_overrides.<name>.output_schema" type="object">
  The tool's output shape.
</ParamField>

<ParamField body="mcp.<name>.tool_overrides.<name>.output_example" type="object">
  An example of the tool's output.
</ParamField>
