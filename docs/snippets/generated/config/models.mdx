<ParamField body="models" type="object">
  Model roles, as `[models.<role>]`. Standard roles are `default`, `chat`, `drafter`, `planner`, `outliner`, `solver`, `repair`, `judge`, `decider` and `image`; any other name is a custom role.
</ParamField>

<ParamField body="models.<name>.type" type="enum<string>">
  What the role's model does. Custom roles default to `chat`; standard roles have a fixed type (`decider` is `decision`, `image` is `image`, the rest are `chat`). The role's provider must serve this type.

  Available options: `chat`, `decision`, `image`
</ParamField>

<ParamField body="models.<name>.provider" type="string" required>
  A name from `[providers]`.
</ParamField>

<ParamField body="models.<name>.model" type="string" required>
  The provider's model id.
</ParamField>

<ParamField body="models.<name>.temperature" type="number">
  Sampling temperature.
</ParamField>

<ParamField body="models.<name>.description" type="string">
  What the role's model is good for, shown to drafters beside the role's name. Standard roles have a built-in one that this replaces; a custom role needs one to be offered through a prompt tool's or `builtin__generate_image`'s `model` input.
</ParamField>

<ParamField body="models.<name>.context_window" type="integer">
  The model's context window in tokens.
</ParamField>

<ParamField body="models.<name>.fallbacks.[].model" type="string" required>

</ParamField>

<ParamField body="models.<name>.fallbacks.[].temperature" type="number">
